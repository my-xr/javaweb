package com.ctts.common;

import io.swagger.v3.oas.annotations.media.Schema;
import lombok.Data;

import java.io.Serializable;

/**
 * 统一响应结果
 *
 * @param <T> 数据类型
 */
@Data
@Schema(description = "统一响应结果")
public class Result<T> implements Serializable {

    @Schema(description = "状态码", example = "200")
    private Integer code;

    @Schema(description = "提示信息", example = "操作成功")
    private String message;

    @Schema(description = "响应数据")
    private T data;

    @Schema(description = "时间戳", example = "1719000000000")
    private Long timestamp;

    public Result() {
        this.timestamp = System.currentTimeMillis();
    }

    /** 成功（无数据） */
    public static <T> Result<T> success() {
        Result<T> result = new Result<>();
        result.setCode(ResultCode.SUCCESS.getCode());
        result.setMessage(ResultCode.SUCCESS.getMessage());
        return result;
    }

    /** 成功（带数据） */
    public static <T> Result<T> success(T data) {
        Result<T> result = new Result<>();
        result.setCode(ResultCode.SUCCESS.getCode());
        result.setMessage(ResultCode.SUCCESS.getMessage());
        result.setData(data);
        return result;
    }

    /** 成功（自定义消息+数据） */
    public static <T> Result<T> success(String message, T data) {
        Result<T> result = new Result<>();
        result.setCode(ResultCode.SUCCESS.getCode());
        result.setMessage(message);
        result.setData(data);
        return result;
    }

    /** 失败（自定义状态码+消息） */
    public static <T> Result<T> error(Integer code, String message) {
        Result<T> result = new Result<>();
        result.setCode(code);
        result.setMessage(message);
        return result;
    }

    /** 失败（使用ResultCode） */
    public static <T> Result<T> error(ResultCode resultCode) {
        Result<T> result = new Result<>();
        result.setCode(resultCode.getCode());
        result.setMessage(resultCode.getMessage());
        return result;
    }

    /** 失败（使用ResultCode+自定义消息） */
    public static <T> Result<T> error(ResultCode resultCode, String message) {
        Result<T> result = new Result<>();
        result.setCode(resultCode.getCode());
        result.setMessage(message);
        return result;
    }
}

1.2 ResultCode状态码枚举

文件：src/main/java/com/ctts/common/ResultCode.java

package com.ctts.common;

import lombok.Getter;

/**
 * 系统状态码枚举
 * 规范：
 *  - 200   成功
 *  - 400   参数错误
 *  - 401   未认证
 *  - 403   无权限
 *  - 404   资源不存在
 *  - 405   请求方法不支持
 *  - 409   业务冲突
 *  - 500   系统异常
 */
@Getter
public enum ResultCode {

    SUCCESS(200, "操作成功"),
    PARAM_ERROR(400, "参数错误"),
    PARAM_VALIDATE_FAILED(400, "参数校验失败"),
    UNAUTHORIZED(401, "未登录或登录已过期"),
    FORBIDDEN(403, "无操作权限"),
    NOT_FOUND(404, "资源不存在"),
    METHOD_NOT_ALLOWED(405, "请求方法不支持"),
    BUSINESS_CONFLICT(409, "业务冲突"),
    SYSTEM_ERROR(500, "系统繁忙，请稍后重试");

    private final Integer code;
    private final String message;

    ResultCode(Integer code, String message) {
        this.code = code;
        this.message = message;
    }
}

package com.ctts.common;

import lombok.Getter;

/**
 * 业务异常
 */
@Getter
public class BusinessException extends RuntimeException {

    private final Integer code;

    public BusinessException(String message) {
        super(message);
        this.code = ResultCode.BUSINESS_CONFLICT.getCode();
    }

    public BusinessException(Integer code, String message) {
        super(message);
        this.code = code;
    }

    public BusinessException(ResultCode resultCode) {
        super(resultCode.getMessage());
        this.code = resultCode.getCode();
    }

    public BusinessException(ResultCode resultCode, String message) {
        super(message);
        this.code = resultCode.getCode();
    }
}

二、所有DTO校验注解（步骤1）

位置：com.ctts.dto 包下。步骤1的实践内容——为所有DTO添加JSR303校验注解。 校验注解速查：

  • @NotBlank：字符串非null且非空白（仅CharSequence）
  • @NotNull：非null（任意类型）
  • @NotEmpty：非null且非空（String/Collection/Map/Array）
  • @Size：长度/大小范围
  • @Pattern：正则匹配
  • @Email：邮箱格式
  • @Min/@Max：数值范围
  • @Past/@Future：过去/未来时间

package com.ctts.common;

import jakarta.validation.ConstraintViolation;
import jakarta.validation.ConstraintViolationException;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpStatus;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.validation.BindException;
import org.springframework.validation.FieldError;
import org.springframework.web.HttpRequestMethodNotSupportedException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.MissingServletRequestParameterException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.ResponseStatus;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.method.annotation.MethodArgumentTypeMismatchException;

import java.util.stream.Collectors;

/**
 * 全局异常处理器
 * 处理10类异常：
 *  1. BusinessException           业务异常
 *  2. MethodArgumentNotValidException  @RequestBody参数校验失败（@Valid + @NotBlank等）
 *  3. ConstraintViolationException     @RequestParam/@PathVariable参数校验失败（@Validated + @NotBlank等）
 *  4. BindException                    表单绑定校验失败
 *  5. MissingServletRequestParameterException  缺少必填请求参数
 *  6. MethodArgumentTypeMismatchException      参数类型不匹配
 *  7. HttpMessageNotReadableException          请求体不可读（JSON格式错误）
 *  8. HttpRequestMethodNotSupportedException   请求方法不支持
 *  9. NullPointerException                     空指针
 * 10. Exception                                其他未捕获异常（兜底）
 */
@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler {

    /** 1. 业务异常 */
    @ExceptionHandler(BusinessException.class)
    public Result<Void> handleBusinessException(BusinessException e) {
        log.warn("业务异常：code={}, message={}", e.getCode(), e.getMessage());
        return Result.error(e.getCode(), e.getMessage());

public class ApplyController {

    private final ApplyService applyService;

    @PostMapping
    @Operation(summary = "提交入队申请", description = "用户向指定团队提交入队申请")
    public Result<Void> apply(@Valid @RequestBody ApplyDTO dto) {
        log.info("提交入队申请：teamId={}, userId={}", dto.getTeamId(), dto.getUserId());
        applyService.apply(dto);
        log.info("入队申请提交成功：teamId={}, userId={}", dto.getTeamId(), dto.getUserId());
        return Result.success();
    }

    @PutMapping("/{id}/approve")
    @Operation(summary = "批准申请", description = "队长批准指定申请，申请人自动加入团队")
    public Result<Void> approve(
            @Parameter(description = "申请ID", required = true, example = "1")
            @PathVariable @NotNull(message = "申请ID不能为空") Long id) {
        log.info("批准申请：id={}", id);
        boolean ok = applyService.approve(id);
        if (!ok) {
            log.warn("批准申请失败：id={}", id);
            throw new BusinessException(ResultCode.SYSTEM_ERROR, "批准申请失败");
        }
        log.info("批准申请成功：id={}", id);
        return Result.success();
    }

    @PutMapping("/{id}/reject")
    @Operation(summary = "拒绝申请", description = "队长拒绝指定申请")
    public Result<Void> reject(
            @Parameter(description = "申请ID", required = true, example = "1")
            @PathVariable @NotNull(message = "申请ID不能为空") Long id) {
        log.info("拒绝申请：id={}", id);
        boolean ok = applyService.reject(id);
        if (!ok) {
            log.warn("拒绝申请失败：id={}", id);
            throw new BusinessException(ResultCode.SYSTEM_ERROR, "拒绝申请失败");
        }
        log.info("拒绝申请成功：id={}", id);
        return Result.success();
    }

    @GetMapping("/{id}")
    @Operation(summary = "查询申请详情", description = "根据申请ID查询申请信息")
    public Result<Apply> getById(
            @Parameter(description = "申请ID", required = true, example = "1")
            @PathVariable @NotNull(message = "申请ID不能为空") Long id) {
        log.info("查询申请详情：id={}", id);
        Apply apply = applyService.getById(id);
        if (apply == null) {
            log.warn("申请不存在：id={}", id);
            throw new BusinessException(ResultCode.NOT_FOUND, "申请不存在");
        }
        return Result.success(apply);
    }

    @GetMapping
    @Operation(summary = "查询申请列表", description = "支持按团队、状态、申请人筛选")
    public Result<List<Apply>> list(
            @Parameter(description = "团队ID", example = "1")
            @RequestParam(required = false) Long teamId,
            @Parameter(description = "申请状态（0待审批 1已通过 2已拒绝）", example = "0")
            @RequestParam(required = false) Integer status,
            @Parameter(description = "申请人ID", example = "10")
            @RequestParam(required = false) Long userId) {
        log.info("查询申请列表：teamId={}, status={}, userId={}", teamId, status, userId);
        List<Apply> applies = applyService.list(teamId, status, userId);
        return Result.success(applies);
    }
}
