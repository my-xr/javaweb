CREATE TABLE `project` (

CREATE TABLE `task` (
  `id` bigint NOT NULL AUTO_INCREMENT,

3. 统一响应与异常处理

3.1 统一响应 Result

// common/Result.java
package com.ctts.common;

import lombok.Data;

@Data
public class Result<T> {
    private Integer code;
    private String message;
    private T data;
    private Long timestamp;

    private Result() {
        this.timestamp = System.currentTimeMillis();
    }

    public static <T> Result<T> success(T data) {
        Result<T> r = new Result<>();
        r.setCode(ResultCode.SUCCESS.getCode());
        r.setMessage(ResultCode.SUCCESS.getMessage());
        r.setData(data);
        return r;
    }

    public static <T> Result<T> success() {
        return success(null);
    }

    public static <T> Result<T> fail(ResultCode rc) {
        Result<T> r = new Result<>();
        r.setCode(rc.getCode());
        r.setMessage(rc.getMessage());
        return r;
    }

    public static <T> Result<T> fail(Integer code, String message) {
        Result<T> r = new Result<>();
        r.setCode(code);
        r.setMessage(message);
        return r;
    }
}

// common/ResultCode.java
package com.ctts.common;

import lombok.AllArgsConstructor;
import lombok.Getter;

@Getter
@AllArgsConstructor
public enum ResultCode {
    SUCCESS(200, "操作成功"),
    PARAM_ERROR(400, "参数错误"),
    UNAUTHORIZED(401, "未登录或token失效"),
    FORBIDDEN(403, "无权限访问"),
    NOT_FOUND(404, "资源不存在"),
    BUSINESS_ERROR(1001, "业务异常"),
    SUBMISSION_DUPLICATE(1002, "该任务已提交且未评审,不可重复提交"),
    PROJECT_STATUS_ERROR(1003, "项目状态不允许此操作"),
    SERVER_ERROR(5000, "服务器异常");

    private final Integer code;
    private final String message;
}

// exception/BusinessException.java
package com.ctts.exception;

import com.ctts.common.ResultCode;
import lombok.Getter;

@Getter
public class BusinessException extends RuntimeException {
    private final Integer code;

    public BusinessException(ResultCode rc) {
        super(rc.getMessage());
        this.code = rc.getCode();
    }

    public BusinessException(Integer code, String message) {
        super(message);
        this.code = code;
    }
}

// exception/GlobalExceptionHandler.java
package com.ctts.exception;

import com.ctts.common.Result;
import com.ctts.common.ResultCode;
import jakarta.validation.ConstraintViolationException;
import lombok.extern.slf4j.Slf4j;
import org.springframework.validation.BindException;
import org.springframework.validation.FieldError;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(BusinessException.class)
    public Result<Void> handleBusiness(BusinessException e) {
        log.warn("业务异常: {}", e.getMessage());
        return Result.fail(e.getCode(), e.getMessage());
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public Result<Void> handleValid(MethodArgumentNotValidException e) {
        FieldError fe = e.getBindingResult().getFieldError();
        String msg = fe != null ? fe.getDefaultMessage() : "参数校验失败";
        log.warn("参数校验失败: {}", msg);
        return Result.fail(ResultCode.PARAM_ERROR.getCode(), msg);
    }

    @ExceptionHandler(BindException.class)
    public Result<Void> handleBind(BindException e) {
        String msg = e.getBindingResult().getFieldError() != null
                ? e.getBindingResult().getFieldError().getDefaultMessage() : "参数绑定失败";
        return Result.fail(ResultCode.PARAM_ERROR.getCode(), msg);
    }

    @ExceptionHandler(ConstraintViolationException.class)
    public Result<Void> handleConstraint(ConstraintViolationException e) {
        String msg = e.getConstraintViolations().iterator().next().getMessage();
        return Result.fail(ResultCode.PARAM_ERROR.getCode(), msg);
    }

    @ExceptionHandler(Exception.class)
    public Result<Void> handleException(Exception e) {
        log.error("系统异常", e);
        return Result.fail(ResultCode.SERVER_ERROR);
    }
}

// service/impl/ProjectServiceImpl.java
package com.ctts.service.impl;

import com.baomidou.mybatisplus.extension.service.impl.ServiceImpl;
import com.ctts.dto.teacher.ProjectCreateDTO;
import com.ctts.entity.Project;
import com.ctts.exception.BusinessException;
import com.ctts.common.ResultCode;
import com.ctts.mapper.ProjectMapper;
import com.ctts.service.ProjectService;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
@RequiredArgsConstructor
public class ProjectServiceImpl extends ServiceImpl<ProjectMapper, Project>
        implements ProjectService {

    @Override
    @Transactional(rollbackFor = Exception.class)
    public Project createProject(ProjectCreateDTO dto, Long teacherId) {
        if (dto.getEndTime() != null && dto.getStartTime() != null
                && dto.getEndTime().isBefore(dto.getStartTime())) {
            throw new BusinessException(ResultCode.PARAM_ERROR.getCode(),
                    "结束时间不能早于开始时间");
        }
        Project project = new Project();
        project.setTitle(dto.getTitle());
        project.setDescription(dto.getDescription());
        project.setTeacherId(teacherId);
        project.setStatus("DRAFT");
        project.setStartTime(dto.getStartTime());
        project.setEndTime(dto.getEndTime());
        save(project);
        return project;
    }
}

package com.ctts.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.extension.service.impl.ServiceImpl;
import com.ctts.dto.student.GradeVO;
import com.ctts.dto.student.SubmissionFeedbackVO;
import com.ctts.dto.teacher.ReviewDTO;
import com.ctts.dto.teacher.SubmissionReviewVO;
import com.ctts.entity.Submission;
import com.ctts.entity.Task;
import com.ctts.exception.BusinessException;
import com.ctts.common.ResultCode;
import com.ctts.mapper.SubmissionMapper;
import com.ctts.mapper.TaskMapper;
import com.ctts.service.SubmissionService;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDateTime;
import java.util.List;

@Service
@RequiredArgsConstructor
public class SubmissionServiceImpl extends ServiceImpl<SubmissionMapper, Submission>
        implements SubmissionService {

    private final SubmissionMapper submissionMapper;
    private final TaskMapper taskMapper;

    @Override
    public List<SubmissionReviewVO> listByTask(Long taskId) {
        return submissionMapper.selectVOByTask(taskId);
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void review(Long submissionId, ReviewDTO dto) {
        Submission sub = getById(submissionId);
        if (sub == null) {
            throw new BusinessException(ResultCode.NOT_FOUND);
        }
        Task task = taskMapper.selectById(sub.getTaskId());
        if (task != null && dto.getScore() > task.getMaxScore()) {
            throw new BusinessException(ResultCode.PARAM_ERROR.getCode(),
                    "得分不能超过满分" + task.getMaxScore());
        }
        sub.setScore(dto.getScore());
        sub.setComment(dto.getComment());
        sub.setStatus("GRADED");
        sub.setReviewTime(LocalDateTime.now());
        updateById(sub);
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public Submission submit(Long taskId, Long studentId, String content) {
        // 唯一性校验:同一学生同一任务,若存在PENDING(待评审)提交,不允许重复提交
        Long exists = lambdaQuery()
                .eq(Submission::getTaskId, taskId)
                .eq(Submission::getStudentId, studentId)
                .eq(Submission::getStatus, "PENDING")
                .count();
        if (exists > 0) {
            throw new BusinessException(ResultCode.SUBMISSION_DUPLICATE);
        }
        Task task = taskMapper.selectById(taskId);
        if (task == null) {
            throw new BusinessException(ResultCode.NOT_FOUND.getCode(), "任务不存在");
        }
        if (LocalDateTime.now().isAfter(task.getDeadline())) {
            throw new BusinessException(ResultCode.BUSINESS_ERROR.getCode(), "任务已过截止时间");
        }
        Submission sub = new Submission();
        sub.setTaskId(taskId);
        sub.setStudentId(studentId);
        sub.setContent(content);
        sub.setStatus("PENDING");
        sub.setSubmitTime(LocalDateTime.now());
        save(sub);
        return sub;
    }

    @Override
    public SubmissionFeedbackVO getFeedback(Long submissionId, Long studentId) {
        SubmissionFeedbackVO vo = submissionMapper.selectFeedback(submissionId, studentId);
        if (vo == null) {
            throw new BusinessException(ResultCode.NOT_FOUND);
        }
        return vo;
    }

    @Override
    public List<GradeVO> listGrades(Long studentId) {
        return submissionMapper.selectGrades(studentId);
    }
}

    public User createUser(AdminUserCreateDTO dto) {
        Long exists = lambdaQuery().eq(User::getUsername, dto.getUsername()).count();

            throw new BusinessException(ResultCode.BUSINESS_ERROR.getCode(), "用户名已存在");

        user.setRole(dto.getRole());
        user.setCollege(dto.getCollege());
        user.setClassName(dto.getClassName());
        user.setStatus(1);
        save(user);
        return user;
    }

        User user = getById(uid);
        if (user == null) {
            throw new BusinessException(ResultCode.NOT_FOUND);
        }
        if (dto.getRealName() != null) user.setRealName(dto.getRealName());
        if (dto.getRole() != null) user.setRole(dto.getRole());
        if (dto.getCollege() != null) user.setCollege(dto.getCollege());
        if (dto.getClassName() != null) user.setClassName(dto.getClassName());
        if (dto.getStatus() != null) user.setStatus(dto.getStatus());
        updateById(user);
    }

9. 实践操作步骤指导

步骤 1：准备项目骨架（10 分钟）

目标：确保项目可启动，数据库连接正常。

操作：

  1. 导入 ctts 项目到 IDEA。
  2. 执行 ctts.sql 创建数据库和测试数据。
  3. 修改 application.yml 数据库连接信息。
  4. 启动 CttsApplication，访问 http://localhost:8080/doc.html 查看 Knife4j 文档。

验证：

  • 控制台输出 Started CttsApplication。
  • Knife4j 文档页面可访问，显示已有接口。

步骤 2：教师端 - 创建实训项目接口（15 分钟）

操作：

  1. 新建 ProjectCreateDTO，添加校验注解。
  2. 新建 ProjectController，编写 create 方法。
  3. 新建 ProjectService 和 ProjectServiceImpl，实现 createProject。
  4. 确认 ProjectMapper 继承 BaseMapper<Project>。

验证（Knife4j）：

  • 请求：POST /api/teacher/projects，body：{"title":"Java Web实训","description":"..."}
  • 期望响应：{"code":200,"message":"操作成功","data":{"id":1,"title":"Java Web实训",...}}
  • 异常用例：title 为空 → {"code":400,"message":"项目标题不能为空"}

步骤 3：教师端 - 发布任务接口（10 分钟）

操作：

  1. 新建 TaskCreateDTO（含 @Future 校验）。
  2. 在 TaskController 编写 publish 方法。
  3. 在 TaskServiceImpl 实现 publishTask，校验项目存在性和状态。

  • 异常用例：deadline 早于当前时间 → {"code":400,"message":"截止时间必须晚于当前时间"}

步骤 4：教师端 - 查看项目下任务列表（多表关联）（15 分钟）

  3. 在 TaskMapper.xml 编写多表关联 SQL（task LEFT JOIN project LEFT JOIN submission 聚合）。
  4. 在 TaskController 编写 listByProject 方法。

验证：

  • 请求：GET /api/teacher/projects/1/tasks
  • 期望：返回 List<TaskVO>，每条含 projectTitle 和 submissionCount。
  • 检查：当某任务无提交时，submissionCount 为 0（非 null）。

步骤 5：教师端 - 查看学生提交情况 + 评审接口（15 分钟）

11. 常见问题排查表

序号 问题现象 可能原因 排查方法 解决方案
1 接口返回 401 未登录 未携带 token 或 token 失效 检查请求头 Authorization 重新登录获取 token
2 接口返回 403 无权限 当前用户角色不匹配 检查 token 中 role 字段 使用正确角色的账号
3 参数校验不生效 未加 @Valid 或 @Validated 检查 Controller 方法参数注解 添加 @Valid
4 多表关联查询字段为 null resultMap 字段映射错误 检查 XML 中 column 与 SQL 别名 保持 column 与别名一致
5 submissionCount 为 null LEFT JOIN 子查询无数据 检查 SQL 是否用 IFNULL 使用 IFNULL(cnt, 0)
6 评审后 status 未变 未调用 updateById 或事务回滚 检查 Service 是否加 @Transactional 加事务注解
7 重复提交未被拦截 唯一性查询条件错误 检查 lambdaQuery 条件 确保 status=PENDING 条件
8 密码明文存储 未使用 BCrypt 加密 检查 UserService 使用 passwordEncoder.encode
9 Knife4j 不显示接口 未加 @Tag 或未配置扫描 检查 Controller 注解和配置 添加 @Tag 注解
10 项目状态无法变更 DTO 校验注解正则错误 检查 @Pattern 正则 修正正则表达式
11 全局异常未捕获 异常类型未覆盖 检查 @ExceptionHandler 补充对应异常处理
12 分页查询不生效 未配置 MyBatis-Plus 拦截器 检查 MybatisPlusConfig 添加 PaginationInnerInterceptor
13 时间字段格式错误 未配置 Jackson 时区 检查 application.yml 配置 spring.jackson.time-zone=GMT+8
14 @Future 校验不生效 时间格式不正确 检查请求体时间格式 使用 ISO 标准格式 yyyy-MM-dd'T'HH:mm:ss
15 多表关联 SQL 报错 表名或字段名错误 查看控制台 SQL 日志 核对数据库表结构
