1.3 WebMvcConfig —— 注册拦截器

package com.ctts.config;

import com.ctts.interceptor.JwtInterceptor;
import com.ctts.interceptor.RoleInterceptor;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.context.annotation.Configuration;
import org.springframework.web.servlet.config.annotation.InterceptorRegistry;
import org.springframework.web.servlet.config.annotation.WebMvcConfigurer;

/**
 * Web MVC 配置：注册 JWT 与角色拦截器
 * 注意：拦截器执行顺序按 registration 顺序，先 JwtInterceptor 后 RoleInterceptor
 */
@Configuration
public class WebMvcConfig implements WebMvcConfigurer {

    @Autowired
    private JwtInterceptor jwtInterceptor;

    @Autowired
    private RoleInterceptor roleInterceptor;

    @Override
    public void addInterceptors(InterceptorRegistry registry) {
        // 1. JWT 认证拦截器（先执行）
        registry.addInterceptor(jwtInterceptor)
                .addPathPatterns("/**")
                .excludePathPatterns(
                        "/auth/login",          // 登录放行
                        "/auth/register",       // 注册放行
                        "/doc.html",            // Knife4j 文档
                        "/swagger-ui/**",
                        "/v3/api-docs/**",
                        "/webjars/**",
                        "/favicon.ico"
                )
                .order(1);

        // 2. 角色权限拦截器（后执行，依赖 JWT 已注入的用户信息）
        registry.addInterceptor(roleInterceptor)
                .addPathPatterns("/**")
                .excludePathPatterns(
                        "/auth/login",
                        "/auth/register",
                        "/doc.html",
                        "/swagger-ui/**",
                        "/v3/api-docs/**",
                        "/webjars/**",
                        "/favicon.ico"
                )
                .order(2);
    }
}

二、权限控制完整实现

2.1 自定义注解 @RequireRole

package com.ctts.annotation;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;

/**
 * 角色权限注解
 * 标注在 Controller 类或方法上，声明访问所需的角色
 * - 标注在类上：类内所有方法都需要该角色
 * - 标注在方法上：方法级别权限覆盖类级别
 * - 支持多角色（满足任一即可访问）
 */
@Target({ElementType.METHOD, ElementType.TYPE})
@Retention(RetentionPolicy.RUNTIME)
public @interface RequireRole {

    /**
     * 允许访问的角色编码（Role.code），满足任一即可
     */
    String[] value();

    /**
     * 是否要求全部角色都满足（默认 false，即满足任一即可）
     */
    boolean requireAll() default false;
}

2.2 RoleInterceptor —— 角色权限拦截器

package com.ctts.interceptor;

import com.ctts.annotation.RequireRole;
import com.ctts.common.BaseContext;
import com.ctts.exception.PermissionException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;
import org.springframework.web.method.HandlerMethod;
import org.springframework.web.servlet.HandlerInterceptor;

import java.util.Arrays;
import java.util.Set;

/**
 * 角色权限拦截器
 * 工作流程：
 *   1. 获取当前请求目标方法/类的 @RequireRole 注解
 *   2. 从 ThreadLocal 读取 JWT 拦截器注入的当前用户角色
 *   3. 校验角色是否匹配，不匹配抛出 PermissionException
 */
@Slf4j
@Component
public class RoleInterceptor implements HandlerInterceptor {

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        // 只拦截 Controller 方法
        if (!(handler instanceof HandlerMethod handlerMethod)) {
            return true;
        }

        // 优先读取方法上的注解（方法级覆盖类级）
        RequireRole methodAnno = handlerMethod.getMethodAnnotation(RequireRole.class);
        RequireRole classAnno = handlerMethod.getBeanType().getAnnotation(RequireRole.class);
        RequireRole anno = methodAnno != null ? methodAnno : classAnno;

        // 未标注 @RequireRole，默认放行（仅做登录校验）
        if (anno == null) {
            return true;
        }

        // 从 ThreadLocal 获取当前用户角色（由 JwtInterceptor 注入）
        Set<String> currentUserRoles = BaseContext.getCurrentRoles();
        if (currentUserRoles == null || currentUserRoles.isEmpty()) {
            log.warn("[权限拦截] 用户角色为空，URI={}, method={}",
                    request.getRequestURI(), handlerMethod.getMethod().getName());
            throw new PermissionException("用户角色信息缺失，请重新登录");
        }

        String[] requiredRoles = anno.value();
        boolean requireAll = anno.requireAll();

        boolean pass;
        if (requireAll) {
            // 要求全部满足
            pass = Arrays.stream(requiredRoles).allMatch(currentUserRoles::contains);
        } else {
            // 满足任一即可
            pass = Arrays.stream(requiredRoles).anyMatch(currentUserRoles::contains);
        }

        if (!pass) {
            log.warn("[权限拦截] 角色不匹配，URI={}, required={}, current={}",
                    request.getRequestURI(), Arrays.toString(requiredRoles), currentUserRoles);
            throw new PermissionException("无权访问该资源，所需角色：" + Arrays.toString(requiredRoles));
        }

        log.debug("[权限拦截] 校验通过，URI={}, currentRoles={}", request.getRequestURI(), currentUserRoles);
        return true;
    }
}

2.3 JwtInterceptor —— JWT 认证拦截器

package com.ctts.interceptor;

import com.ctts.common.BaseContext;
import com.ctts.exception.AuthException;
import com.ctts.utils.JwtUtil;
import io.jsonwebtoken.Claims;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;
import org.springframework.util.StringUtils;
import org.springframework.web.servlet.HandlerInterceptor;

import java.util.Set;

/**
 * JWT 认证拦截器
 * 职责：校验 Token 有效性，并将用户信息注入 ThreadLocal 供后续使用
 */
@Slf4j
@Component
public class JwtInterceptor implements HandlerInterceptor {

    @Autowired
    private JwtUtil jwtUtil;

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        // 1. 从请求头获取 Token
        String token = request.getHeader(jwtUtil.getHeader());
        if (!StringUtils.hasText(token)) {
            log.warn("[认证拦截] 未携带 Token, URI={}", request.getRequestURI());
            throw new AuthException("未登录或登录已过期，请重新登录");
        }

        // 2. 去除 Bearer 前缀
        if (token.startsWith(jwtUtil.getPrefix())) {
            token = token.substring(jwtUtil.getPrefix().length());
        }

        // 3. 解析与校验 Token
        Claims claims;
        try {
            claims = jwtUtil.parseToken(token);
        } catch (Exception e) {
            log.warn("[认证拦截] Token 解析失败, URI={}, msg={}", request.getRequestURI(), e.getMessage());
            throw new AuthException("Token 无效或已过期，请重新登录");
        }

        // 4. 将用户信息注入 ThreadLocal（供 RoleInterceptor 与 Controller 使用）
        Long userId = claims.get("uid", Long.class);
        String username = claims.get("username", String.class);
        @SuppressWarnings("unchecked")
        Set<String> roles = claims.get("roles", Set.class);

        BaseContext.setCurrentUserId(userId);
        BaseContext.setCurrentUsername(username);
        BaseContext.setCurrentRoles(roles);

        log.debug("[认证拦截] Token 校验通过, userId={}, username={}, roles={}", userId, username, roles);
        return true;
    }

    @Override
    public void afterCompletion(HttpServletRequest request, HttpServletResponse response,
                                Object handler, Exception ex) {
        // 清理 ThreadLocal，防止内存泄漏
        BaseContext.clear();
    }
}

2.4 BaseContext —— ThreadLocal 上下文

package com.ctts.common;

import java.util.Set;

/**
 * 请求上下文：存储当前请求用户信息（基于 ThreadLocal）
 * 必须在拦截器 afterCompletion 中清理，防止线程池复用导致的内存泄漏
 */
public class BaseContext {

    private static final ThreadLocal<Long> TL_USER_ID = new ThreadLocal<>();
    private static final ThreadLocal<String> TL_USERNAME = new ThreadLocal<>();
    private static final ThreadLocal<Set<String>> TL_ROLES = new ThreadLocal<>();

    public static void setCurrentUserId(Long id) { TL_USER_ID.set(id); }
    public static Long getCurrentUserId() { return TL_USER_ID.get(); }

    public static void setCurrentUsername(String name) { TL_USERNAME.set(name); }
    public static String getCurrentUsername() { return TL_USERNAME.get(); }

    public static void setCurrentRoles(Set<String> roles) { TL_ROLES.set(roles); }
    public static Set<String> getCurrentRoles() { return TL_ROLES.get(); }

    public static void clear() {
        TL_USER_ID.remove();
        TL_USERNAME.remove();
        TL_ROLES.remove();
    }
}

2.5 JwtUtil —— JWT 工具类

package com.ctts.utils;

import com.ctts.exception.AuthException;
import io.jsonwebtoken.Claims;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import jakarta.annotation.PostConstruct;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.util.Date;
import java.util.Map;
import java.util.Set;

@Component
public class JwtUtil {

    @Value("${ctts.jwt.secret}")
    private String secret;

    @Value("${ctts.jwt.ttl}")
    private long ttl;

    @Value("${ctts.jwt.header}")
    private String header;

    @Value("${ctts.jwt.prefix}")
    private String prefix;

    private SecretKey key;

    @PostConstruct
    public void init() {
        this.key = Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8));
    }

    /** 生成 JWT */
    public String generateToken(Long userId, String username, Set<String> roles) {
        return Jwts.builder()
                .claims(Map.of("uid", userId, "username", username, "roles", roles))
                .subject(String.valueOf(userId))
                .issuedAt(new Date())
                .expiration(new Date(System.currentTimeMillis() + ttl))
                .signWith(key)
                .compact();
    }

    /** 解析 JWT */
    public Claims parseToken(String token) {
        return Jwts.parser()
                .verifyWith(key)
                .build()
                .parseSignedClaims(token)
                .getPayload();
    }

    public String getHeader() { return header; }
    public String getPrefix() { return prefix; }
}

// ===== 通用业务异常 =====
package com.ctts.exception;

import lombok.Getter;

@Getter
public class BusinessException extends RuntimeException {
    private final Integer code;

    public BusinessException(String message) {
        super(message);
        this.code = 500;
    }

    public BusinessException(Integer code, String message) {
        super(message);
        this.code = code;
    }

public class Result<T> implements Serializable {

    private String msg;

        this.code = code;
        this.msg = msg;
        this.data = data;
        this.timestamp = System.currentTimeMillis();

    public static <T> Result<T> success() {
        return new Result<>(200, "success", null);
    }

    public static <T> Result<T> success(T data) {

    public static <T> Result<T> fail(Integer code, String msg) {
        return new Result<>(code, msg, null);
    }

    public static <T> Result<T> fail(String msg) {

    }
}

3.3 GlobalExceptionHandler 完整实现

package com.ctts.exception;

import com.ctts.common.Result;
import jakarta.validation.ConstraintViolation;
import jakarta.validation.ConstraintViolationException;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpStatus;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.validation.BindException;
import org.springframework.validation.FieldError;
import org.springframework.web.HttpRequestMethodNotSupportedException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.MissingServletRequestParameterException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.ResponseStatus;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.servlet.NoHandlerFoundException;

import java.util.stream.Collectors;

/**
 * 全局异常处理器
 * 处理顺序：具体异常 → 通用异常 → 兜底异常
 * 所有异常均返回统一 Result 结构，避免向前端暴露堆栈信息
 */
@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler {

    // ============ 1. 参数校验异常（@RequestBody @Valid） ============
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public Result<Void> handleValidException(MethodArgumentNotValidException e) {
        String msg = e.getBindingResult().getFieldErrors().stream()
                .map(fe -> fe.getField() + ": " + fe.getDefaultMessage())
                .collect(Collectors.joining("; "));
        log.warn("[参数校验失败] {}", msg);
        return Result.fail(400, msg);
    }

    // ============ 2. 参数校验异常（@RequestParam @Validated） ============
    @ExceptionHandler(ConstraintViolationException.class)
    public Result<Void> handleConstraintViolation(ConstraintViolationException e) {
        String msg = e.getConstraintViolations().stream()
                .map(ConstraintViolation::getMessage)
                .collect(Collectors.joining("; "));
        log.warn("[参数校验失败] {}", msg);
        return Result.fail(400, msg);
    }

    // ============ 3. 表单绑定异常 ============
    @ExceptionHandler(BindException.class)
    public Result<Void> handleBindException(BindException e) {
        FieldError fe = e.getBindingResult().getFieldError();
        String msg = fe != null ? fe.getField() + ": " + fe.getDefaultMessage() : "参数绑定失败";
        log.warn("[参数绑定失败] {}", msg);
        return Result.fail(400, msg);
    }

    // ============ 4. 缺少请求参数 ============
    @ExceptionHandler(MissingServletRequestParameterException.class)
    public Result<Void> handleMissingParam(MissingServletRequestParameterException e) {
        String msg = "缺少必需参数：" + e.getParameterName();
        log.warn("[缺少参数] {}", msg);
        return Result.fail(400, msg);
    }

    // ============ 5. 请求体不可读（JSON 格式错误） ============
    @ExceptionHandler(HttpMessageNotReadableException.class)
    public Result<Void> handleNotReadable(HttpMessageNotReadableException e) {
        log.warn("[请求体解析失败] {}", e.getMessage());
        return Result.fail(400, "请求体格式错误，请检查 JSON");
    }

    // ============ 6. 业务异常 ============
    @ExceptionHandler(BusinessException.class)
    public Result<Void> handleBusiness(BusinessException e) {
        log.warn("[业务异常] code={}, msg={}", e.getCode(), e.getMessage());
        return Result.fail(e.getCode(), e.getMessage());
    }

    // ============ 7. 认证异常（401） ============
    @ExceptionHandler(AuthException.class)
    @ResponseStatus(HttpStatus.UNAUTHORIZED)
    public Result<Void> handleAuth(AuthException e) {
        log.warn("[认证异常] {}", e.getMessage());
        return Result.fail(401, e.getMessage());
    }

    // ============ 8. 权限异常（403） ============
    @ExceptionHandler(PermissionException.class)
    @ResponseStatus(HttpStatus.FORBIDDEN)
    public Result<Void> handlePermission(PermissionException e) {
        log.warn("[权限异常] {}", e.getMessage());
        return Result.fail(403, e.getMessage());
    }

    // ============ 9. 请求方法不支持 ============
    @ExceptionHandler(HttpRequestMethodNotSupportedException.class)
    public Result<Void> handleMethodNotSupported(HttpRequestMethodNotSupportedException e) {
        log.warn("[方法不支持] {}", e.getMessage());
        return Result.fail(405, "请求方法不支持：" + e.getMethod());
    }

    // ============ 10. 资源不存在 ============
    @ExceptionHandler(NoHandlerFoundException.class)
    @ResponseStatus(HttpStatus.NOT_FOUND)
    public Result<Void> handleNotFound(NoHandlerFoundException e) {
        log.warn("[资源不存在] {}", e.getRequestURL());
        return Result.fail(404, "资源不存在");
    }

    // ============ 11. 兜底异常（其他未捕获异常） ============
    @ExceptionHandler(Exception.class)
    @ResponseStatus(HttpStatus.INTERNAL_SERVER_ERROR)
    public Result<Void> handleException(Exception e) {
        log.error("[系统异常] 系统内部错误", e);
        // 不向前端暴露堆栈，只返回友好提示
        return Result.fail(500, "系统繁忙，请稍后重试");
    }
}

说明：异常处理顺序由具体到通用。Spring 会优先匹配最具体的异常处理器，因此 Exception.class 兜底处理器只会在前面所有处理器都不匹配时执行。

七、实践操作步骤指导

步骤 1：整合 JWT 认证

目标：让所有业务接口必须携带合法 Token 才能访问。

操作步骤：

  1. 在 pom.xml 中添加 jjwt 依赖（见 1.1）。
  2. 在 application.yml 中添加 JWT 配置（见 1.2）。
  3. 创建 JwtUtil 工具类（见 2.5）。
  4. 创建 BaseContext ThreadLocal 上下文（见 2.4）。
  5. 创建 JwtInterceptor（见 2.3）。
  6. 创建 WebMvcConfig 注册拦截器，配置放行路径（见 1.3）。

验证方法：

# 1. 启动项目，访问 http://localhost:8080/ctts/doc.html —— 应能正常打开
# 2. 不携带 Token 调用 GET /tasks —— 应返回 401
# 3. 调用 POST /auth/login 获取 Token
# 4. 携带 Token 调用 GET /tasks —— 应返回 200

// WebMvcConfig 中放行路径必须包含 Knife4j，否则文档无法访问

步骤 2：整合角色权限控制

目标：使用 @RequireRole 注解为各 Controller 标注角色，实现声明式权限控制。

操作步骤：

  1. 创建自定义注解 @RequireRole（见 2.1）。
  2. 创建 RoleInterceptor，使用反射读取注解并校验角色（见 2.2）。
  3. 在 WebMvcConfig 中注册 RoleInterceptor，order(2)（在 JWT 之后执行）。
  4. 为各 Controller 添加 @RequireRole 标注（见 2.6）。

验证方法：

# 1. 用学生账号登录，获取 Token
# 2. 用学生 Token 调用 POST /teachers/tasks —— 应返回 403
# 3. 用学生 Token 调用 GET /students/profile —— 应返回 200
# 4. 用管理员账号登录，调用 GET /admin/users —— 应返回 200

关键代码片段：

// RoleInterceptor 中优先读取方法级注解
RequireRole methodAnno = handlerMethod.getMethodAnnotation(RequireRole.class);
RequireRole classAnno = handlerMethod.getBeanType().getAnnotation(RequireRole.class);
RequireRole anno = methodAnno != null ? methodAnno : classAnno;

步骤 3：整合全局异常处理

目标：所有异常统一由 GlobalExceptionHandler 捕获，返回统一 Result 结构。

操作步骤：

  1. 创建自定义异常类：BusinessException、AuthException、PermissionException（见 3.1）。
  2. 创建 GlobalExceptionHandler，使用 @RestControllerAdvice + @ExceptionHandler（见 3.3）。
  3. 在 Service 层抛出 BusinessException 替代直接抛出 RuntimeException。
  4. 在拦截器中抛出 AuthException / PermissionException。

验证方法：

# 1. 提交参数校验失败的请求（如 title 为空）—— 应返回 {"code":400,"msg":"title: 不能为空"}
# 2. 提交不存在的资源 ID —— 应返回 {"code":404,"msg":"资源不存在"}
# 3. 不携带 Token 访问 —— 应返回 {"code":401,"msg":"未登录或登录已过期"}
# 4. 用低权限 Token 访问高权限接口 —— 应返回 {"code":403,"msg":"无权访问"}
# 5. 制造系统异常（如数据库断开）—— 应返回 {"code":500,"msg":"系统繁忙"}

关键代码片段：

// Service 层抛业务异常
if (task == null) {
    throw new BusinessException(404, "任务不存在");
}

// 拦截器抛权限异常
throw new PermissionException("无权访问该资源");

步骤 4：整合统一响应 Result

目标：将所有 Controller 返回值统一改造为 Result<T>。

操作步骤：

  1. 创建 Result<T> 与 ResultCode（见 3.2）。
  2. 批量修改 Controller 方法返回类型为 Result<T>。
  3. 成功返回 Result.success(data) 或 Result.success()。
  4. 失败场景通过抛异常由全局处理器统一处理（不再手动返回 fail）。
  5. 在 Knife4j 中验证文档模型是否正确显示。

验证方法：

# 1. 调用任意接口，确认返回结构为 {code, msg, data, timestamp}
# 2. 打开 Knife4j 文档，确认所有接口响应模型统一为 Result<T>
# 3. 对比改造前后代码，确认 Controller 简洁性提升

改造前后对比（见第四节）：

// 改造前
public Student getStudent(@PathVariable Long id) {
    return studentService.getById(id);
}

// 改造后
public Result<StudentVO> getStudent(@PathVariable Long id) {
    return Result.success(studentService.getById(id));
}

步骤 6：权限漏洞自查

目标：通过实战测试发现并修复权限漏洞，培养安全意识。

操作步骤：

  1. 准备三个角色的测试账号（学生、教师、管理员各1个）。
  2. 分别登录获取三个 Token。
  3. 按"权限漏洞自查清单"（见第六节）逐项测试。
  4. 记录失败项，分析原因，修复漏洞。
  5. 修复后重新测试，直至全部通过。
  6. 填写《权限漏洞自查报告》。

验证方法：

# 1. 越权访问测试：用学生 Token 调用教师/管理员接口，应全部返回 403
# 2. Token 伪造测试：篡改 Token Payload，应全部返回 401
# 3. 参数篡改测试：提交非法参数，应返回 400 且无 SQL 注入风险
# 4. 水平越权测试：用学生 A 的 Token 访问学生 B 的数据，应返回 403 或 404

常见漏洞修复建议：

漏洞类型 修复方案
垂直越权 在 Controller 标注 @RequireRole，拦截器校验角色
水平越权 Service 层根据 BaseContext.getCurrentUserId() 过滤数据，不直接信任前端传入的 userId
Token 篡改 使用 HMAC-SHA256 签名，密钥保密；TTL 不宜过长
SQL 注入 使用 MyBatis-Plus 预编译，禁止拼接 SQL
XSS 入库前过滤或转义特殊字符，输出时使用 @JsonIgnore 或前端转义
参数篡改 @Valid + JSR303 校验，限制字符串长度、数值范围
