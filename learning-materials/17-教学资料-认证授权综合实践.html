三、JwtUtil 工具类（完整）

package com.ctts.util;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import jakarta.annotation.PostConstruct;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.util.Date;
import java.util.HashMap;
import java.util.Map;

/**
 * JWT工具类
 * 负责：生成Token、解析Token、校验Token
 */
@Component
public class JwtUtil {

    @Value("${ctts.jwt.secret}")
    private String secret;

    @Value("${ctts.jwt.expire}")
    private long expireSeconds;

    private SecretKey key;

    @PostConstruct
    public void init() {
        // 密钥长度需 >= 256bit(32字节)，HMAC-SHA256
        this.key = Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8));
    }

    /**
     * 生成JWT Token
     * @param userId   用户ID
     * @param username 用户名
     * @param role     角色
     * @return token字符串
     */
    public String generateToken(Long userId, String username, String role) {
        Map<String, Object> claims = new HashMap<>();
        claims.put("userId", userId);
        claims.put("username", username);
        claims.put("role", role);

        Date now = new Date();
        Date expiry = new Date(now.getTime() + expireSeconds * 1000);

        return Jwts.builder()
                .claims(claims)
                .subject(username)
                .issuedAt(now)
                .expiration(expiry)
                .signWith(key)
                .compact();
    }

    /**
     * 解析Token，返回Claims
     * @throws io.jsonwebtoken.ExpiredJwtException Token过期
     * @throws io.jsonwebtoken.JwtException        Token非法
     */
    public Claims parseToken(String token) {
        return Jwts.parser()
                .verifyWith(key)
                .build()
                .parseSignedClaims(token)
                .getPayload();
    }

    /**
     * 从Token中提取用户ID
     */
    public Long getUserId(String token) {
        Claims claims = parseToken(token);
        return claims.get("userId", Long.class);
    }

    /**
     * 从Token中提取角色
     */
    public String getRole(String token) {
        return parseToken(token).get("role", String.class);
    }

    /**
     * 校验Token是否有效（不抛异常即有效）
     */
    public boolean isValid(String token) {
        try {
            parseToken(token);
            return true;
        } catch (Exception e) {
            return false;
        }
    }
}

四、CurrentUserContext（ThreadLocal上下文）

package com.ctts.context;

import com.ctts.entity.User;

/**
 * 当前登录用户上下文
 * 基于ThreadLocal实现，请求线程内共享用户信息
 *
 * 注意：必须在拦截器afterCompletion中调用remove()
 *       否则线程池复用会导致用户串号
 */
public class CurrentUserContext {

    private static final ThreadLocal<User> CURRENT_USER = new ThreadLocal<>();

    public static void set(User user) {
        CURRENT_USER.set(user);
    }

    public static User get() {
        return CURRENT_USER.get();
    }

    public static Long getUserId() {
        User u = get();
        return u == null ? null : u.getId();
    }

    public static String getRole() {
        User u = get();
        return u == null ? null : u.getRole();
    }

    public static void remove() {
        CURRENT_USER.remove();
    }
}

五、@RequireRole 注解定义

package com.ctts.annotation;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;

/**
 * 角色权限注解
 * 标注在Controller方法或类上，由RoleInterceptor读取并校验
 *
 * 用法示例：
 *   @RequireRole({"TEACHER", "ADMIN"})
 *   @GetMapping("/list")
 *   public Result<?> list() { ... }
 */
@Target({ElementType.METHOD, ElementType.TYPE})
@Retention(RetentionPolicy.RUNTIME)
public @interface RequireRole {
    /**
     * 允许访问的角色列表（满足任一即可）
     */
    String[] value();

    /**
     * 是否要求所有角色都满足（默认false，任一即可）
     */
    boolean requireAll() default false;
}

六、LoginInterceptor（JWT认证拦截器）

package com.ctts.interceptor;

import com.ctts.context.CurrentUserContext;
import com.ctts.entity.User;
import com.ctts.util.JwtUtil;
import io.jsonwebtoken.Claims;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;
import org.springframework.web.servlet.HandlerInterceptor;

/**
 * JWT认证拦截器
 * 职责：校验Token合法性，将用户信息存入ThreadLocal
 * 注意：afterCompletion必须remove，防止线程池复用串号
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class LoginInterceptor implements HandlerInterceptor {

    private final JwtUtil jwtUtil;

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) throws Exception {
        // 1. 从Header获取Token
        String authHeader = request.getHeader("Authorization");
        if (authHeader == null || !authHeader.startsWith("Bearer ")) {
            response.setStatus(401);
            response.setContentType("application/json;charset=UTF-8");
            response.getWriter().write("{\"code\":401,\"message\":\"未携带Token\"}");
            return false;
        }

        String token = authHeader.substring(7);

        // 2. 校验Token
        try {
            Claims claims = jwtUtil.parseToken(token);
            // 3. 构造User对象存入ThreadLocal（不查库，提升性能）
            User user = new User();
            user.setId(claims.get("userId", Long.class));
            user.setUsername(claims.get("username", String.class));
            user.setRole(claims.get("role", String.class));
            CurrentUserContext.set(user);
            log.debug("认证通过, userId={}, role={}", user.getId(), user.getRole());
            return true;
        } catch (io.jsonwebtoken.ExpiredJwtException e) {
            log.warn("Token已过期");
            response.setStatus(401);
            response.setContentType("application/json;charset=UTF-8");
            response.getWriter().write("{\"code\":401,\"message\":\"Token已过期，请重新登录\"}");
            return false;
        } catch (Exception e) {
            log.warn("Token非法: {}", e.getMessage());
            response.setStatus(401);
            response.setContentType("application/json;charset=UTF-8");
            response.getWriter().write("{\"code\":401,\"message\":\"Token非法\"}");
            return false;
        }
    }

    @Override
    public void afterCompletion(HttpServletRequest request, HttpServletResponse response, Object handler, Exception ex) {
        // 关键：请求结束清理ThreadLocal，防止线程池复用导致用户串号
        CurrentUserContext.remove();
    }
}

七、RoleInterceptor（角色权限拦截器）

package com.ctts.interceptor;

import com.ctts.annotation.RequireRole;
import com.ctts.context.CurrentUserContext;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.stereotype.Component;
import org.springframework.web.method.HandlerMethod;
import org.springframework.web.servlet.HandlerInterceptor;

import java.util.Arrays;
import java.util.Set;
import java.util.stream.Collectors;

/**
 * 角色权限拦截器
 * 职责：读取@RequireRole注解，校验当前用户角色是否匹配
 * 注意：必须在LoginInterceptor之后执行（order值更大）
 */
@Component
public class RoleInterceptor implements HandlerInterceptor {

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) throws Exception {
        // 非Controller方法（如静态资源）直接放行
        if (!(handler instanceof HandlerMethod handlerMethod)) {
            return true;
        }

        // 1. 优先取方法上的注解，再取类上的注解
        RequireRole methodAnno = handlerMethod.getMethodAnnotation(RequireRole.class);
        RequireRole classAnno = handlerMethod.getBeanType().getAnnotation(RequireRole.class);
        RequireRole anno = methodAnno != null ? methodAnno : classAnno;

        // 2. 没有注解放行（公开接口）
        if (anno == null) {
            return true;
        }

        // 3. 获取当前用户角色
        String currentRole = CurrentUserContext.getRole();
        if (currentRole == null) {
            response.setStatus(401);
            response.setContentType("application/json;charset=UTF-8");
            response.getWriter().write("{\"code\":401,\"message\":\"未登录\"}");
            return false;
        }

        // 4. 校验角色
        Set<String> requiredRoles = Arrays.stream(anno.value()).collect(Collectors.toSet());
        boolean matched;
        if (anno.requireAll()) {
            matched = requiredRoles.contains(currentRole) && requiredRoles.size() == 1;
        } else {
            matched = requiredRoles.contains(currentRole);
        }

        if (!matched) {
            response.setStatus(403);
            response.setContentType("application/json;charset=UTF-8");
            response.getWriter().write("{\"code\":403,\"message\":\"权限不足，需要角色: " + requiredRoles + "\"}");
            return false;
        }
        return true;
    }
}

八、WebMvcConfig（拦截器注册）

package com.ctts.config;

import com.ctts.interceptor.LoginInterceptor;
import com.ctts.interceptor.RoleInterceptor;
import lombok.RequiredArgsConstructor;
import org.springframework.context.annotation.Configuration;
import org.springframework.web.servlet.config.annotation.InterceptorRegistry;
import org.springframework.web.servlet.config.annotation.WebMvcConfigurer;

/**
 * WebMvc配置
 * 注册两个拦截器：
 *   1. LoginInterceptor  (order=1)  负责认证
 *   2. RoleInterceptor   (order=2)  负责授权
 *
 * 执行顺序：LoginInterceptor先执行，校验通过后RoleInterceptor才能读到角色
 */
@Configuration
@RequiredArgsConstructor
public class WebMvcConfig implements WebMvcConfigurer {

    private final LoginInterceptor loginInterceptor;
    private final RoleInterceptor roleInterceptor;

    @Override
    public void addInterceptors(InterceptorRegistry registry) {
        // 1. JWT认证拦截器——拦截所有/api/**路径
        registry.addInterceptor(loginInterceptor)
                .addPathPatterns("/api/**")
                .excludePathPatterns(
                        "/api/auth/register",
                        "/api/auth/login"
                )
                .order(1);

        // 2. 角色权限拦截器——同样拦截/api/**，order值更大表示后执行
        registry.addInterceptor(roleInterceptor)
                .addPathPatterns("/api/**")
                .excludePathPatterns(
                        "/api/auth/register",
                        "/api/auth/login"
                )
                .order(2);
    }
}

9.2 ServiceImpl实现

package com.ctts.service.impl;

import com.ctts.common.BusinessException;
import com.ctts.context.CurrentUserContext;
import com.ctts.entity.User;
import com.ctts.mapper.UserMapper;
import com.ctts.service.UserService;
import com.ctts.util.JwtUtil;
import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.stereotype.Service;

import java.util.HashMap;
import java.util.Map;

@Slf4j
@Service
@RequiredArgsConstructor
public class UserServiceImpl implements UserService {

    private final UserMapper userMapper;
    private final JwtUtil jwtUtil;
    // BCrypt编码器（单例复用，每次加密内部会自动生成不同盐值）
    private final BCryptPasswordEncoder passwordEncoder = new BCryptPasswordEncoder();

    @Override
    public void register(String username, String password, String role, String realName, String phone) {
        // 1. 校验用户名唯一
        Long count = userMapper.selectCount(
                new LambdaQueryWrapper<User>().eq(User::getUsername, username)
        );
        if (count > 0) {
            throw BusinessException.of(400, "用户名已存在");
        }

        // 2. 校验角色合法性
        if (!"STUDENT".equals(role) && !"TEACHER".equals(role) && !"ADMIN".equals(role)) {
            throw BusinessException.of(400, "角色非法，仅支持 STUDENT/TEACHER/ADMIN");
        }

        // 3. BCrypt加密密码（每次盐值不同）
        String encodedPassword = passwordEncoder.encode(password);

        // 4. 构造实体入库
        User user = new User();
        user.setUsername(username);
        user.setPassword(encodedPassword);
        user.setRole(role);
        user.setRealName(realName);
        user.setPhone(phone);
        user.setStatus(1);
        userMapper.insert(user);
        log.info("注册成功, username={}, role={}", username, role);
    }

    @Override
    public Map<String, Object> login(String username, String password) {
        // 1. 根据用户名查库
        User user = userMapper.selectOne(
                new LambdaQueryWrapper<User>().eq(User::getUsername, username)
        );
        if (user == null) {
            throw BusinessException.of(401, "用户名或密码错误");
        }

        // 2. 校验账号状态
        if (user.getStatus() != null && user.getStatus() == 0) {
            throw BusinessException.of(403, "账号已被禁用");
        }

        // 3. BCrypt校验密码（matches方法会自动比对盐值）
        if (!passwordEncoder.matches(password, user.getPassword())) {
            throw BusinessException.of(401, "用户名或密码错误");
        }

        // 4. 生成JWT Token
        String token = jwtUtil.generateToken(user.getId(), user.getUsername(), user.getRole());

        // 5. 组装返回（密码不返回）
        Map<String, Object> result = new HashMap<>();
        result.put("token", token);
        Map<String, Object> userInfo = new HashMap<>();
        userInfo.put("id", user.getId());
        userInfo.put("username", user.getUsername());
        userInfo.put("role", user.getRole());
        userInfo.put("realName", user.getRealName());
        result.put("userInfo", userInfo);
        log.info("登录成功, username={}", username);
        return result;
    }

    @Override
    public User getCurrentUserInfo() {
        // 从ThreadLocal取当前用户ID（避免业务层直接依赖Token）
        Long userId = CurrentUserContext.getUserId();
        if (userId == null) {
            throw BusinessException.of(401, "未登录");
        }
        // 查库获取最新信息
        User user = userMapper.selectById(userId);
        if (user == null) {
            throw BusinessException.of(404, "用户不存在");
        }
        // 脱敏：清空密码字段
        user.setPassword(null);
        return user;
    }

    @Override
    public void changePassword(String oldPassword, String newPassword) {
        // 1. 获取当前登录用户
        Long userId = CurrentUserContext.getUserId();
        if (userId == null) {
            throw BusinessException.of(401, "未登录");
        }
        User user = userMapper.selectById(userId);
        if (user == null) {
            throw BusinessException.of(404, "用户不存在");
        }

        // 2. 校验旧密码
        if (!passwordEncoder.matches(oldPassword, user.getPassword())) {
            throw BusinessException.of(400, "旧密码错误");
        }

        // 3. 校验新密码与旧密码不能相同
        if (passwordEncoder.matches(newPassword, user.getPassword())) {
            throw BusinessException.of(400, "新密码不能与旧密码相同");
        }

        // 4. 加密新密码并更新
        String encodedNewPassword = passwordEncoder.encode(newPassword);
        User update = new User();
        update.setId(userId);
        update.setPassword(encodedNewPassword);
        userMapper.updateById(update);
        log.info("修改密码成功, userId={}", userId);
    }
}

package com.ctts.controller;

import com.ctts.annotation.RequireRole;
import com.ctts.common.Result;
import com.ctts.entity.User;
import com.ctts.service.UserService;
import lombok.Data;
import lombok.RequiredArgsConstructor;
import org.springframework.web.bind.annotation.*;

import java.util.Map;

/**
 * 认证授权控制器
 * 路径约定：
 *   /api/auth/**   认证相关（注册/登录，无需Token）
 *   /api/user/**   用户相关（需Token）
 *   /api/teacher/** 教师专属（需TEACHER/ADMIN角色）
 *   /api/admin/**   管理员专属（需ADMIN角色）
 */
@RestController
@RequestMapping("/api")
@RequiredArgsConstructor
public class AuthController {

    private final UserService userService;

    // ==================== 公开接口（无需Token） ====================

    /**
     * 用户注册
     * POST /api/auth/register
     */
    @PostMapping("/auth/register")
    public Result<Void> register(@RequestBody RegisterDTO dto) {
        userService.register(dto.getUsername(), dto.getPassword(),
                dto.getRole(), dto.getRealName(), dto.getPhone());
        return Result.success(null);
    }

    /**
     * 用户登录
     * POST /api/auth/login
     */
    @PostMapping("/auth/login")
    public Result<Map<String, Object>> login(@RequestBody LoginDTO dto) {
        Map<String, Object> data = userService.login(dto.getUsername(), dto.getPassword());
        return Result.success(data);
    }

    // ==================== 需登录接口（任意角色） ====================

    /**
     * 获取当前登录用户信息
     * GET /api/user/current
     */
    @GetMapping("/user/current")
    public Result<User> currentUser() {
        return Result.success(userService.getCurrentUserInfo());
    }

    /**
     * 修改密码
     * PUT /api/user/password
     */
    @PutMapping("/user/password")
    public Result<Void> changePassword(@RequestBody ChangePasswordDTO dto) {
        userService.changePassword(dto.getOldPassword(), dto.getNewPassword());
        return Result.success(null);
    }

    // ==================== 角色专属接口（演示权限控制） ====================

    /**
     * 教师接口——仅TEACHER和ADMIN可访问
     * GET /api/teacher/task-list
     */
    @RequireRole({"TEACHER", "ADMIN"})
    @GetMapping("/teacher/task-list")
    public Result<String> teacherTaskList() {
        return Result.success("教师任务列表（仅教师/管理员可见）");
    }

    /**
     * 管理员接口——仅ADMIN可访问
     * GET /api/admin/user-list
     */
    @RequireRole({"ADMIN"})
    @GetMapping("/admin/user-list")
    public Result<String> adminUserList() {
        return Result.success("用户管理列表（仅管理员可见）");
    }

    /**
     * 学生接口——仅STUDENT可访问
     * GET /api/student/my-task
     */
    @RequireRole({"STUDENT"})
    @GetMapping("/student/my-task")
    public Result<String> studentMyTask() {
        return Result.success("我的实训任务（仅学生可见）");
    }

    // ==================== DTO定义 ====================

    @Data
    public static class RegisterDTO {
        private String username;
        private String password;
        private String role;
        private String realName;
        private String phone;
    }

    @Data
    public static class LoginDTO {
        private String username;
        private String password;
    }

    @Data
    public static class ChangePasswordDTO {
        private String oldPassword;
        private String newPassword;
    }
}

十五、课堂踩坑复盘要点

踩坑1：Token过期处理

  • 现象：用户登录2小时后访问接口突然返回401。
  • 原因：JWT的exp字段到期，解析时抛ExpiredJwtException。
  • 方案：前端在收到401时跳转登录页；课后拓展实现滑动过期（每次访问刷新Token）。

踩坑2：越权访问

  • 现象：学生Token能访问管理员接口。
  • 原因：忘记加@RequireRole注解，或RoleInterceptor未注册。
  • 排查：用学生Token逐个访问各角色接口，确保403正确返回。
  • 思政：越权访问是线上常见事故，必须每个敏感接口都加权限注解。

踩坑3：ThreadLocal泄漏

  • 现象：A用户登录后，B用户偶尔看到A的数据。
  • 原因：LoginInterceptor.afterCompletion未调用remove()，Tomcat线程池复用线程时残留旧用户。
  • 复现：用JMeter并发模拟，未remove时必现。
  • 解决：afterCompletion中必须调用CurrentUserContext.remove()。
  • 思政：这是典型的"测试通过但生产事故"案例，体现工程严谨性。

十六、拓展练习（课后）

  1. Token滑动续期：在LoginInterceptor中检测Token剩余有效期，小于30分钟则签发新Token通过Header返回。
  2. Redis黑名单注销：实现注销接口，将Token加入Redis黑名单，LoginInterceptor校验是否在黑名单中。
  3. 操作审计日志：写一个AuditInterceptor，对/admin/**接口记录操作人、时间、IP、接口路径到数据库。
  4. 接口限流：用Guava RateLimiter对登录接口做限流（5次/分钟），防止暴力破解。
